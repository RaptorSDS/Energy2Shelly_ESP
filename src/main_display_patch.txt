3. FUNKTION ZUM ABRUFEN DER ZEIT
=============================================================================
Position: VOR der WifiManagerSetup() Funktion (Zeile ~830)

--- CODE EINFÜGEN ---
#ifdef ESP32_S3_GEEK_DISPLAY
String getTimeString() {
  struct tm timeinfo;
  if(!getLocalTime(&timeinfo)){
    return "--:--:--";
  }
  char timeStr[9];
  strftime(timeStr, sizeof(timeStr), "%H:%M:%S", &timeinfo);
  return String(timeStr);
}
#endif

--- ENDE ---
